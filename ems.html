ขั้นตอนการติดตั้งสคริปต์ใน Google Sheets (ทำครั้งเดียว):

  1. เปิด Google Sheets ของคุณ -> ไปที่เมนู ส่วนขยาย (Extensions) -> Apps Script
  2. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดด้านล่างนี้ไปวางแทนที่
  3. กดปุ่ม ทำให้ใช้ได้อย่างเป็นทางการ (Deploy) -> เลือก การทําการแจ้งใหม่ (New deployment)
  4. ตั้งค่าประเภทเป็น เว็บแอป (Web app)
  5. สำคัญมาก: ตั้งค่า "เรียกใช้ในฐานะ (Execute as)" = ฉัน (Me) และ "ผู้มีสิทธิ์เข้าถึง (Who has access)" = ทุกคน (Anyone)
  6. กดคัดลอก URL ของ Web App มาวางในช่องด้านบนแล้วกด "บันทึกการตั้งค่า"

      headers = ["วัน-เวลา", "สถานะผู้ตอบ", "ตำบล", "ประเภทผู้ป่วย", "ข้อ 1 ความรวดเร็ว", "ข้อ 2 การดูแล", "ข้อ 3 การสื่อสาร", "ข้อ 4 อุปกรณ์", "ข้อ 5 ภาพรวม", "คะแนนเฉลี่ย", "ข้อเสนอแนะ"];
      rowData = [data.date, data.role, data.subdistrict, data.patientType, data.scores[0], data.scores[1], data.scores[2], data.scores[3], data.scores[4], data.avg, data.comment];
    } else if (data.formType === "f2") {
      sheetName = "2.เครือข่าย EMS";
      headers = ["วัน-เวลา", "หน่วยงาน", "ข้อ 1 ประสานงาน", "ข้อ 2 วิชาการ/ทรัพยากร", "ข้อ 3 การรับช่วงดูแล", "ข้อ 4 สัมพันธภาพ", "ข้อ 5 ภาพรวม", "คะแนนเฉลี่ย", "ข้อเสนอแนะ"];
      rowData = [data.date, data.org, data.scores[0], data.scores[1], data.scores[2], data.scores[3], data.scores[4], data.avg, data.comment];
    } else if (data.formType === "f3") {
      sheetName = "3.บุคลากรภายใน";
      headers = ["วัน-เวลา", "ตำแหน่ง", "ข้อ 1 ตารางเวร", "ข้อ 2 ความปลอดภัย", "ข้อ 3 การสนับสนุน", "ข้อ 4 พัฒนาศักยภาพ", "ข้อ 5 บรรยากาศ", "คะแนนเฉลี่ย", "ข้อเสนอแนะ"];
      rowData = [data.date, data.pos, data.scores[0], data.scores[1], data.scores[2], data.scores[3], data.scores[4], data.avg, data.comment];
    } else if (data.formType === "f4") {
      sheetName = "4.หน่วยงานรับส่งต่อ";
      headers = ["วัน-เวลา", "ชื่อโรงพยาบาล", "วิชาชีพ", "ข้อ 1 แจ้งล่วงหน้า", "ข้อ 2 บันทึกข้อมูล", "ข้อ 3 คุณภาพรักษา", "ข้อ 4 ความพร้อมอุปกรณ์", "ข้อ 5 ภาพรวม", "คะแนนเฉลี่ย", "ข้อเสนอแนะ"];
      rowData = [data.date, data.hospital, data.prof, data.scores[0], data.scores[1], data.scores[2], data.scores[3], data.scores[4], data.avg, data.comment];
    } else if (data.formType === "complaint") {
      sheetName = "5.ข้อร้องเรียน";
      headers = ["รหัส", "วัน-เวลาเกิดเหตุ", "ผู้แจ้ง", "เบอร์โทร", "ประเภทผู้ร้อง", "หมวดหมู่", "รายละเอียด", "การแก้ไขเบื้องต้น", "สถานะ"];
      rowData = [data.id, data.date, data.reporter, data.phone, data.type, data.category, data.details, data.action, data.status];
    } else if (data.formType === "test") {
      sheetName = "ทดสอบการเชื่อมต่อ";
      headers = ["วัน-เวลา", "สถานะ"];
      rowData = [data.date, "เชื่อมต่อสำเร็จ"];
    }

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#dcfce7");
    }

    sheet.appendRow(rowData);
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
